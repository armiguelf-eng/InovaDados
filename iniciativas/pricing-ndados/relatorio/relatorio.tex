% --- fontes da identidade ----------------------------------------------------
% Host Grotesk nos títulos, Nunito Sans no texto corrido. Os dois arquivos são
% variable fonts (eixo `wght`), então cada peso é pedido pelo eixo em vez de
% por um arquivo estático — daí o `RawFeature={axis={...}}`.
\usepackage{fontspec}
\newcommand{\pastaFontes}{assets/fontes/}   % copiada para assets/ pelo gerador

% `BoldFont`/`ItalicFont` precisam nomear o arquivo mesmo sendo o mesmo da
% versão normal: sem isso fontspec não registra a forma /b/n e o \textbf cai
% silenciosamente no peso regular.
\setmainfont{NunitoSans-Variable.ttf}[
  Path               = \pastaFontes,
  UprightFeatures    = {RawFeature={axis={wght=400}}},
  BoldFont           = NunitoSans-Variable.ttf,
  BoldFeatures       = {RawFeature={axis={wght=700}}},
  ItalicFont         = NunitoSans-Italic-Variable.ttf,
  ItalicFeatures     = {RawFeature={axis={wght=400}}},
  BoldItalicFont     = NunitoSans-Italic-Variable.ttf,
  BoldItalicFeatures = {RawFeature={axis={wght=700}}},
]

\newfontfamily\fonteTitulo{HostGrotesk-Variable.ttf}[
  Path               = \pastaFontes,
  UprightFeatures    = {RawFeature={axis={wght=600}}},
  BoldFont           = HostGrotesk-Variable.ttf,
  BoldFeatures       = {RawFeature={axis={wght=700}}},
  ItalicFont         = HostGrotesk-Italic-Variable.ttf,
  ItalicFeatures     = {RawFeature={axis={wght=600}}},
  BoldItalicFont     = HostGrotesk-Italic-Variable.ttf,
  BoldItalicFeatures = {RawFeature={axis={wght=700}}},
]
